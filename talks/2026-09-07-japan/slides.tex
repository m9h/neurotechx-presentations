\begin{frame}{Different interfaces need different partners}
  \begin{columns}[T,onlytextwidth]
    \column{.47\textwidth}\NTXCard{For an implanted BCI project}{Which clinical team and intended users shape the design?\par\medskip Who supports implantation, follow-up, and long-term use?}
    \column{.47\textwidth}\NTXCard{For a non-invasive BCI project}{Where and by whom will it be used?\par\medskip How will setup, signal quality, comfort, and repeat use be evaluated?}
  \end{columns}
  \NTXTakeaway{Choose a use case before choosing a collaboration model.}
  \note{[7:30--9:00] Frame these as questions for project teams, not a complete clinical or regulatory checklist. Non-invasive does not automatically mean consumer use or low risk. Intended use matters. This meeting can connect projects to the appropriate specialists rather than prescribing one route for all technologies.}
\end{frame}

\begin{frame}{At the AI interface, agree on the test}
  \begin{enumerate}
    \setlength\itemsep{4mm}
    \item What does the model add over a simple baseline?
    \item Does the result hold across sessions and people?
    \item What happens when the model is uncertain or wrong?
    \item Does the complete system help the intended user?
  \end{enumerate}
  \NTXTakeaway{A shared evaluation plan makes collaboration concrete.}
  \note{[9:00--10:30] These are proposed evaluation questions. Distinguish held-out evaluation from training performance, and offline decoding from useful interaction. A benchmark can support algorithm comparison; prospective evaluation with intended users asks a further question.}
\end{frame}

  \NTXSource{https://github.com/NeuroTechX/moabb}{MOABB: project description and evaluation example}

\begin{frame}{Give people a path from interest to contribution}
  \begin{columns}[T,onlytextwidth]
    \column{.31\textwidth}\NTXCard{Enter}{A welcoming introductory session.\par\medskip Explain the problem and prerequisites.}
    \column{.31\textwidth}\NTXCard{Practice}{A guided workshop using a reproducible example.\par\medskip Pair different disciplines.}
    \column{.31\textwidth}\NTXCard{Contribute}{A scoped issue and an available mentor.\par\medskip Publish what the team learned.}
  \end{columns}
  \NTXTakeaway{Measure whether people return and finish useful work.}
  \note{[12:30--14:00] This is a proposed community program. A meetup is an entry point; a supported project gives people a reason to return. Technical documentation, translation, design, and user research can be valuable contributions alongside code. Make mentorship an explicit responsibility.}
\end{frame}

\begin{frame}{Make international collaboration specific}
  \begin{columns}[T,onlytextwidth]
    \column{.47\textwidth}\NTXCard{Start with one shared artifact}{A benchmark replication\par A bilingual technical guide\par A jointly reviewed project brief}
    \column{.47\textwidth}\NTXCard{Give it a working agreement}{One owner on each side\par A regular meeting time\par A review date and success criterion}
  \end{columns}
  \NTXTakeaway{Translate the work so that the next team can participate.}
  \note{[14:00--15:30] English documentation supports international participation; Japanese learning materials support local access. Choose a small artifact with reciprocal value. \end{frame}

\begin{frame}{A proposed first 90 days}
  \begin{columns}[T,onlytextwidth]
    \column{.31\textwidth}\NTXCard{Days 1--30 / Map}{Identify willing participants.\par\medskip Choose one problem and two project leads.}
    \column{.31\textwidth}\NTXCard{Days 31--60 / Build}{Run a joint workshop.\par\medskip Produce a reproducible first artifact.}
    \column{.31\textwidth}\NTXCard{Days 61--90 / Review}{Demonstrate the work.\par\medskip Publish lessons and decide the next step.}
  \end{columns}
  \NTXTakeaway{Leave today with an owner and a date for the next conversation.}
  \note{[17:00--19:00] Explicitly call this a proposal for discussion. Start with public data or other material that participants are entitled to share. The goal is to test whether a collaboration can work. A useful review can conclude that the project should change direction or stop.}
\end{frame}

\begin{frame}{Measure what the collaboration makes possible}

    \textcolor{NTXCyan}{Question} & \textcolor{NTXCyan}{Evidence to bring to the review}\\
    \midrule
    Did people become contributors? & Completed tasks and returning participants.\\[4mm]
    Can another team use the work? & An independent reproduction or documented reuse.\\[4mm]
    Did the project move forward? & A receiving partner and an agreed next milestone.\\[4mm]

    \bottomrule
  \end{tabular}
